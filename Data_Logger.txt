class MillisUnwrapper:
    """Convert the board's uint32 millis counter into monotonic seconds.

    Unsigned subtraction naturally handles the roughly 49.7-day wrap. A
    backward jump smaller than half the uint32 range is treated as a board
    reset and starts a new segment without making plotted time go backward.
    """

    def __init__(self):
        self.last_ms = None
        self.elapsed_ms = 0
        self.reset_count = 0

    def update(self, ms):
        ms &= 0xFFFFFFFF
        reset = False
        if self.last_ms is not None:
            delta = (ms - self.last_ms) & 0xFFFFFFFF
            if delta <= 0x7FFFFFFF:
                self.elapsed_ms += delta
            else:
                self.reset_count += 1
                reset = True
        self.last_ms = ms
        return self.elapsed_ms / 1000.0, reset


def open_unique_csv_pair(directory, base):
    """Create two recording files without ever truncating an old session."""
    os.makedirs(directory, exist_ok=True)
    for index in range(10000):
        suffix = "" if index == 0 else f"_{index:02d}"
        data_path = os.path.join(directory, base + suffix + ".csv")
        peak_path = os.path.join(directory, base + suffix + "_peaks.csv")
        data_file = None
        try:
            data_file = open(data_path, "x", newline="")
            try:
                peak_file = open(peak_path, "x", newline="")
            except FileExistsError:
                data_file.close()
                os.unlink(data_path)
                continue
            except Exception:
                data_file.close()
                os.unlink(data_path)
                raise
            return data_path, peak_path, data_file, peak_file
        except FileExistsError:
            continue
    raise OSError("could not allocate a unique recording filename")


def list_ports():
    ports = list(serial.tools.list_ports.comports())
    if not ports:
        print("No serial ports found.")
    for p in ports:
        print(f"  {p.device}  —  {p.description}")
    return ports


def pick_port():
    ports = list_ports()
    if not ports:
        print("Plug in the board and try again, or pass --port explicitly.")
        sys.exit(1)
    if len(ports) == 1:
        print(f"Using the only port available: {ports[0].device}")
        return ports[0].device
    print()
    for i, p in enumerate(ports):
        print(f"  [{i}] {p.device}  —  {p.description}")
    while True:
        choice = input(f"Select port [0-{len(ports) - 1}]: ").strip()
        try:
            idx = int(choice)
            if 0 <= idx < len(ports):
                return ports[idx].device
        except ValueError:
            pass
        print("Invalid selection, try again.")

def config_fingerprint(manifest):
    fields = {key: manifest[key] for key in ('analysis_policy', 'detector_config',
              'avg_window', 'expected_expiration_sign', 'flow_reference')}
    encoded = json.dumps(fields, sort_keys=True, separators=(',', ':'), allow_nan=False)
    return hashlib.sha256(encoded.encode('ascii')).hexdigest()


def uint32(value):
    result = int(value)
    if not 0 <= result <= 0xffffffff:
        raise ValueError('uint32 out of range')
    return result


def parse_wire(line):
    p = line.split(',')
    protocol = {'DATA': 1, 'DATA2': 2}.get(p[0])
    if protocol is None or len(p) != (37 if protocol == 1 else 44):
        raise ValueError('unknown DATA schema/width')
    def finite(i):
        value = float(p[i])
        if not math.isfinite(value):
            raise ValueError('nonfinite field')
        return value
    def flag(i):
        if p[i] not in ('0', '1'):
            raise ValueError('invalid flag')
        return p[i] == '1'
    def age(i, valid):
        value = int(p[i])
        if value < -1 or value > 0x7fffffff or (valid and value < 0):
            raise ValueError('invalid age')
        return value
    flow_ok = flag(4)
    pok = [flag(i) for i in range(14, 21)]
    phase, preset, dac = int(p[34]), int(p[35]), int(p[28])
    if phase not in (-1, 0, 1) or preset not in (-2, -1, 0, 1, 2, 3) or not 0 <= dac <= 4095:
        raise ValueError('invalid phase/preset/DAC')
    if (preset >= 0 and phase == -1) or (preset < 0 and phase != -1):
        raise ValueError('phase/preset mismatch')
    rid = p[37] if protocol == 2 else None
    if protocol == 2 and not re.fullmatch('[0-9a-f]{16}', rid):
        raise ValueError('invalid run ID')
    return dict(protocol=protocol, seq=uint32(p[1]), ms=uint32(p[2]),
                flow_raw=finite(3), flow_ok=flow_ok, flow_age_us=age(5, flow_ok),
                pa=[finite(i) for i in range(6, 13)], temp=finite(13), pok=pok,
                pa_age_us=[age(i, ok) for i, ok in zip(range(21, 28), pok)],
                cmd_dac=dac, feedback_dac=finite(29), cmd_pct=finite(30), fdbk_pct=finite(31),
                target_flow_slm=finite(32), target_peak_slm=finite(33), phase=phase, preset=preset,
                clamped=flag(36), run_id=rid,
                **{key: uint32(p[i]) if protocol == 2 else None for key, i in
                   zip(('scan_id', 'row_us', 'cycle_id', 'scan_start_us', 'missed_slots_total', 'tx_dropped_total'), range(38, 44))})


def encode_cell(value):
    if isinstance(value, (list, tuple)):
        return json.dumps(value, separators=(',', ':'), allow_nan=True)
    if value is None:
        return ''
    if isinstance(value, bool):
        return int(value)
    if isinstance(value, float):
        return format(value, '.12g')
    return value


def sample_record(sample, line):
    return dict(wire_line=line, **{key: encode_cell(value) for key, value in sample._asdict().items()})


def event_record(event):
    result = {key: encode_cell(value) for key, value in event.items() if key != 'sample'}
    result.update(seq=event['sample'].seq, cycle_id=event['sample'].cycle_id, phase=event['sample'].phase)
    return result


def open_unique_session(directory, base):
    """Reserve all three related paths; clean up only paths this attempt owns."""
    os.makedirs(directory, exist_ok=True)
    for index in range(10000):
        name = base + (f'_{index:02d}' if index else '')
        paths = [os.path.join(directory, name + suffix) for suffix in ('.csv', '_peaks.csv', '_session.json')]
        files = []
        try:
            for path in paths:
                files.append(open(path, 'x', newline=''))
            return paths, files
        except Exception as exc:
            for path, stream in zip(paths, files):
                stream.close()
                os.unlink(path)
            if not isinstance(exc, FileExistsError):
                raise
    raise OSError('could not allocate unique recording names')


class SerialLink:
    """Own the serial reader, protocol-2 run lifecycle, plots and recordings.

    The reader incrementally frames bounded ASCII records. Owned motion uses
    correlated START/PING/STOP frames; legacy data remains display-only.
    Run state precedes recording state in lock order. ACK waits hold neither.
    """

    def __init__(self, port, baud, avg_window, buffer_seconds=180.0,
                 sample_hz=100.0, peak_config=None,
                 run_seconds=DEFAULT_RUN_SECONDS, gas_temperature_c=None,
                 absolute_pressure_mbar=None, expected_expiration_sign=None,
                 serial_instance=None, start_reader=True):
        self.ser = serial_instance if serial_instance is not None else serial.Serial(port, baud, timeout=0.2, write_timeout=1.0)
        self._serial_write_lock = threading.Lock()
        self._heartbeat_interval = 0.5
        self._last_heartbeat = 0.0
        self.connection_error = None
        self.connected = True
        self._ready_event = threading.Event()
        self.avg_window = max(1, avg_window)
        self.peak_config = peak_config or PeakDetectionConfig()
        self._flow_hist = deque(maxlen=self.avg_window)
        self._flow_sum = 0.0  # running sum: O(1) average instead of O(w)

        maxlen = int(buffer_seconds * sample_hz)
        self.lock = threading.Lock()
        self.t = deque(maxlen=maxlen)
        self.flow_raw = deque(maxlen=maxlen)
        self.flow_avg = deque(maxlen=maxlen)
        self.flow_ok = deque(maxlen=maxlen)
        self.flow_age_us = deque(maxlen=maxlen)
        self.psens_pa = [deque(maxlen=maxlen) for _ in range(NUM_PSENS)]
        self.psens_temp_avg = deque(maxlen=maxlen)
        self.psens_ok = [deque(maxlen=maxlen) for _ in range(NUM_PSENS)]
        self.psens_age_us = [deque(maxlen=maxlen) for _ in range(NUM_PSENS)]
        self.cmd_dac = deque(maxlen=maxlen)
        self.feedback_dac = deque(maxlen=maxlen)
        self.cmd_pct = deque(maxlen=maxlen)
        self.fdbk_pct = deque(maxlen=maxlen)
        self.target_flow_slm = deque(maxlen=maxlen)
        self.target_peak_slm = deque(maxlen=maxlen)
        self.phase = deque(maxlen=maxlen)
        self.preset = deque(maxlen=maxlen)
        self.clamped = deque(maxlen=maxlen)

        self._clock = MillisUnwrapper()
        self._stop = threading.Event()
        self.bad_lines = 0
        self.dropped_rows = 0
        self._last_seq = None

        # One snapshot is taken per animation frame and shared by all three
        # figures. Previously each figure independently copied the complete
        # buffer, duplicating the largest redraw cost three times per frame.
        self._snap_cache = None
        self._snap_stamp = 0.0
        self._snap_ttl = 0.04

        # Recording state — guarded separately from the plot-buffer lock so
        # a slow disk write can never stall the live plots.
        self._record_lock = threading.Lock()
        self._record_file = None
        self._record_writer = None
        self._record_path = None
        self._peak_file = None
        self._peak_writer = None
        self._peak_path = None
        self._detector = None
        self._rows_written = 0
        self._run_bad_lines = 0
        self._run_dropped_rows = 0
        self._run_first_t_s = None
        self._run_last_t_s = None
        self._last_preset = None  # most recent preset seen in a DATA row
        self._record_expected_preset = None
        self._record_started = False

        # Run state uses a generation token so a cancelled Timer callback can
        # never stop or centre a newer run that starts at the same boundary.
        self.run_seconds = run_seconds
        self._run_lock = threading.Lock()
        self._run_timer = None
        self._run_start_monotonic = None
        self._run_generation = 0

        standard_flow(1.0, gas_temperature_c, absolute_pressure_mbar)
        self.gas_temperature_c = gas_temperature_c
        self.absolute_pressure_mbar = absolute_pressure_mbar
        self.expected_expiration_sign = expected_expiration_sign
        self.state = 'IDLE'
        self.run_id = None
        self.mode = None
        self.protocol_version = None
        self.metadata = None
        self._meta_pending = None
        self._meta_build = None
        self._meta_invalid = False
        self._heartbeat_enabled = False
        self._started_event = threading.Event()
        self._ended_event = threading.Event()
        self._rx = bytearray()
        self._rx_discard = False
        self._manifest_path = None
        self.manifest = None
        self.orientation_mismatches = 0
        self._reset_analysis_state()

        self._thread = threading.Thread(target=self._reader_loop, daemon=True)
        if start_reader:
            self._thread.start()

    def _write_bytes_locked(self, data, timeout=None):
        """Guard timeout configuration and writes; caller holds the write lock."""
        try:
            previous_timeout = getattr(self.ser, 'write_timeout', None)
            if timeout is not None:
                self.ser.write_timeout = timeout
            try:
                if self.ser.write(data) != len(data):
                    raise OSError('short serial write')
            finally:
                if timeout is not None:
                    self.ser.write_timeout = previous_timeout
            return True
        except (serial.SerialException, OSError, ValueError) as exc:
            self._heartbeat_enabled = False
            self.connection_error = str(exc)
            self.connected = False
            if not self._stop.is_set():
                print(f'-> transport fault: {exc}; stop unconfirmed')
            return False

    def send_command(self, ch):
        if ch not in COMMANDS:
            return False
        if ch == '?':
            with self._run_lock:
                if self.state in ('STARTING', 'RUNNING', 'STOPPING'):
                    print('-> finish the active run before renegotiating readiness')
                    return False
                self._ready_event.clear()
                self.protocol_version = None
                self.metadata = None
                self._meta_pending = None
                self._meta_build = None
                self._meta_invalid = False
                with self._serial_write_lock:
                    return self._write_bytes_locked(b'\n?')
        if ch in PRESET_KEYS:
            return self.begin_timed_run(int(ch)-1, ch)
        if ch in MOTION_OVERRIDE_KEYS:
            if self.run_in_progress() and not self.stop_run():
                return False
            return self.begin_timed_run(-2, ch)
        confirmed = self.state != 'FAULTED'
        if ch in SAFETY_OVERRIDE_KEYS and self.run_in_progress():
            confirmed = self.stop_run()
        # An unconfirmed protocol stop must not suppress an explicit operator
        # override. Resynchronize any partial frame before the legacy command.
        data = (('\n' if ch in SAFETY_OVERRIDE_KEYS else '') + ch).encode('ascii')
        with self._run_lock:
            owned_id = self.run_id if self.state in ('STARTING', 'RUNNING', 'STOPPING') else None
            with self._serial_write_lock:
                written = self._write_bytes_locked(data)
        if not written and owned_id is not None:
            self._fault('transport_fault', expected_run_id=owned_id)
        return written and confirmed

    def _send_heartbeat(self):
        with self._serial_write_lock:
            now = time.monotonic()
            if not self._heartbeat_enabled or now-self._last_heartbeat < self._heartbeat_interval:
                return True
            ok = self._write_bytes_locked(f'@PING,{self.run_id}\n'.encode('ascii'))
            if ok:
                self._last_heartbeat = now
            return ok

    def wait_until_ready(self, timeout):
        if not self.send_command('?'):
            return False
        return self._ready_event.wait(timeout)

    def _reader_loop(self):
        try:
            while not self._stop.is_set():
                if not self._send_heartbeat():
                    self._fault('transport_fault')
                    return
                try:
                    raw = self.ser.read(256)
                except (serial.SerialException, OSError) as exc:
                    if not self._stop.is_set():
                        self._fault('transport_fault', exc)
                    return
                if raw:
                    self.feed_bytes(raw)
        except Exception as exc:
            # Unexpected analysis/format failures must retain a traceback and
            # disable host ownership; never silently kill only the reader.
            if not self._stop.is_set():
                traceback.print_exc()
                self._fault('analysis_fault', exc)
        finally:
            self.connected = False

    def feed_bytes(self, raw):
        """Incremental bounded framing; timeout never turns fragments into rows."""
        for byte in raw:
            if byte == 10:
                if not self._rx_discard and self._rx:
                    try:
                        line = self._rx.decode('ascii').rstrip('\r')
                    except UnicodeDecodeError:
                        self._count_bad_line()
                    else:
                        self._handle_line(line)
                self._rx.clear()
                self._rx_discard = False
            elif not self._rx_discard:
                if len(self._rx) >= 1024:
                    self._rx.clear()
                    self._rx_discard = True
                    self._count_bad_line()
                else:
                    self._rx.append(byte)

    def _handle_line(self, line):
        if line.startswith('DATA'):
            self._handle_data_line(line)
        elif line.startswith('# '):
            self._handle_marker(line)

    def _handle_marker(self, line):
        try:
            self._process_marker(line)
        except RecordingFault as exc:
            self._fault(exc.kind, exc.cause)
        except OSError as exc:
            self._fault('storage_fault', exc)
        except Exception as exc:
            traceback.print_exc()
            self._fault('analysis_fault', exc)

    def _process_marker(self, line):
        parts = line[2:].split(',')
        kind = parts[0]
        if kind == 'BOOT':
            if self.run_in_progress():
                self._fault('board_reset')
            self.protocol_version = None
            self.metadata = None
            self._ready_event.clear()
        elif parts == ['PROTOCOL', '2']:
            if self.run_in_progress():
                self._fault('unexpected_handshake')
            self.protocol_version = 2
        elif kind == 'META_BEGIN' and len(parts) == 3 and parts[1] == '2':
            self._meta_build = parts[2]
            self._meta_pending = {}
            self._meta_invalid = False
        elif kind == 'META' and len(parts) == 4 and parts[1] == '2':
            if self._meta_pending is None or parts[2] in self._meta_pending:
                self._meta_invalid = True
            elif not all(re.fullmatch(r'[A-Za-z0-9_.+\-]+', x) for x in parts[2:]):
                self._meta_invalid = True
            else:
                self._meta_pending[parts[2]] = parts[3]
        elif kind == 'META_END' and len(parts) == 3:
            if (parts[1] == '2' and parts[2] == self._meta_build and
                    not self._meta_invalid and self._meta_pending is not None and
                    set(self._meta_pending) == META_KEYS):
                self.metadata = dict(self._meta_pending, build_id=self._meta_build)
            else:
                self.metadata = None
            self._meta_pending = None
        elif parts == ['READY_QUERY']:
            # Legacy is display-only. Timed acquisition checks v2 + metadata.
            if (self.protocol_version is None and self.state != 'FAULTED') or self.metadata is not None:
                self._ready_event.set()
                with self._run_lock:
                    if self.state == 'FAULTED' and self.metadata is not None:
                        self.state = 'IDLE'
        elif kind == 'ERROR':
            print(f'-> firmware error: {line[2:]}')
        elif kind == 'STARTED' and len(parts) == 6 and parts[1] == '2':
            try:
                start_ms, start_us = uint32(parts[4]), uint32(parts[5])
            except ValueError:
                self._fault('invalid_start_ack')
                return
            with self._run_lock:
                if self.state != 'STARTING' or parts[2] != self.run_id or parts[3] != self.mode:
                    return
                with self._record_lock:
                    self._reset_analysis_state(start_ms, start_us)
                    self._record_started = True
                    if self.manifest is not None:
                        self.manifest['start_ack'] = {'millis': start_ms, 'micros': start_us}
                        self._save_manifest()
                self.state = 'RUNNING'
                self._run_start_monotonic = time.monotonic()
                timer = threading.Timer(self.run_seconds, self._on_run_timeout, args=(self._run_generation,))
                timer.daemon = True
                self._run_timer = timer
                timer.start()
                self._started_event.set()
        elif kind == 'ENDED' and len(parts) == 6 and parts[1] == '2':
            try:
                final_seq = -1 if parts[3] == '-1' else uint32(parts[3])
                dropped = uint32(parts[5])
                if parts[4] not in ('host_stop', 'watchdog', 'manual_override', 'transport_fault'):
                    raise ValueError('unknown termination reason')
            except ValueError:
                self._fault('invalid_end_ack')
                return
            with self._run_lock:
                if self.state not in ('RUNNING', 'STOPPING', 'STARTING') or parts[2] != self.run_id:
                    return
                with self._serial_write_lock:
                    self._heartbeat_enabled = False
                self._cancel_run_timer_locked()
                with self._record_lock:
                    incomplete = (final_seq != (self._last_seq if self._last_seq is not None else -1) or
                                  self._run_dropped_rows > 0 or self._run_bad_lines > 0 or dropped > 0)
                    if self.manifest is not None:
                        self.manifest['end_ack'] = dict(final_enqueued_seq=final_seq, reason=parts[4], tx_dropped_total=dropped)
                    self._finalize_locked(parts[4], incomplete)
                self.state = 'IDLE'
                self._ended_event.set()
                if parts[4] in ('host_stop', 'watchdog', 'transport_fault'):
                    print('-> firmware hold confirmed; recording finalized')
                else:
                    print('-> firmware session ended by manual override; recording finalized')

    def _reset_analysis_state(self, start_ms=0, start_us=0):
        self._flow_hist.clear()
        self._flow_sum = 0.0
        self._last_seq = None
        self._last_sample = None
        self._clock = MillisUnwrapper()
        self._clock.update(start_ms)
        self._last_row_us = start_us
        self._elapsed_row_us = 0
        self._segment_id = 0
        self._phase_first = None
        self._phase_last = None
        self._completed_phases = set()
        with self.lock:
            for name in ('t', 'flow_raw', 'flow_avg', 'flow_ok', 'flow_age_us',
                         'psens_temp_avg', 'cmd_dac', 'feedback_dac', 'cmd_pct',
                         'fdbk_pct', 'target_flow_slm', 'target_peak_slm', 'phase',
                         'preset', 'clamped'):
                getattr(self, name).clear()
            for name in ('psens_pa', 'psens_ok', 'psens_age_us'):
                for series in getattr(self, name):
                    series.clear()

    def _call_detector_locked(self, method, *args):
        """Detach a failed detector before teardown can retry its callback.

        Caller holds the recording lock. Lifecycle teardown is performed by
        the caller after releasing that lock to preserve lock ordering.
        """
        if self._detector is None:
            return
        try:
            return getattr(self._detector, method)(*args)
        except RecordingFault:
            self._detector = None
            raise
        except OSError as exc:
            self._detector = None
            raise RecordingFault('storage_fault', exc) from exc
        except Exception as exc:
            self._detector = None
            traceback.print_exc()
            raise RecordingFault('analysis_fault', exc) from exc

    def _break_locked(self, reason, before_seq=None):
        self._flow_hist.clear()
        self._flow_sum = 0.0
        self._segment_id += 1
        if self.manifest is not None:
            self.manifest['boundaries'].append(dict(after_seq=self._last_seq, before_seq=before_seq, reason=reason))
        self._call_detector_locked('break_continuity', reason)

    def _handle_data_line(self, line):
        try:
            wire = parse_wire(line)
        except ValueError:
            self._count_bad_line()
            return
        fault = None
        with self._record_lock:
            active = self._record_started and self.state in ('RUNNING', 'STOPPING')
            if self.state == 'STARTING':
                return
            if active and (wire['run_id'] != self.run_id or wire['preset'] != self._record_expected_preset):
                fault = 'unexpected_session'
            elif active and wire['protocol'] != 2:
                fault = 'legacy_row_in_v2_session'
            if fault is None:
                try:
                    sample = self._sample_from_wire(wire, active)
                except BoardTimeReset:
                    fault = 'board_time_reset' if active else None
                    if not active:
                        self._reset_analysis_state(wire['ms'], wire['row_us'] or 0)
                    sample = None
                except RecordingFault as exc:
                    fault = exc.kind
                    self.connection_error = str(exc.cause)
                    sample = None
                except Exception:
                    traceback.print_exc()
                    fault = 'analysis_fault'
                    self._detector = None
                    sample = None
                if sample is not None:
                    self._plot_sample(sample)
                    if active and self._record_writer is not None:
                        try:
                            self._record_writer.writerow(sample_record(sample, line))
                            self._rows_written += 1
                            if self._rows_written % 100 == 0:
                                self._record_file.flush()
                                self._save_manifest()
                        except OSError as exc:
                            fault = 'storage_fault'
                            self.connection_error = str(exc)
                        except Exception:
                            traceback.print_exc()
                            fault = 'format_fault'
                        if fault is None:
                            try:
                                self._call_detector_locked('push', sample)
                            except RecordingFault as exc:
                                fault = exc.kind
                                self.connection_error = str(exc.cause)
        if fault:
            self._fault(fault)

    def _sample_from_wire(self, w, active):
        elapsed_s, reset = self._clock.update(w['ms'])
        if reset:
            raise BoardTimeReset('millis reset')
        if w['protocol'] == 2:
            delta = (w['row_us'] - self._last_row_us) & 0xffffffff
            if delta > 0x7fffffff:
                raise BoardTimeReset('micros reset')
            self._elapsed_row_us += delta
            self._last_row_us = w['row_us']
            if active and abs(self._elapsed_row_us/1e6-elapsed_s) > .003:
                raise BoardTimeReset('millis/micros disagree')
            row_time_us = self._elapsed_row_us
            flow_time_us = row_time_us-w['flow_age_us'] if w['flow_age_us'] >= 0 else row_time_us
            t_s = flow_time_us/1e6
        else:
            row_time_us = int(elapsed_s*1e6)
            flow_time_us = row_time_us-w['flow_age_us'] if w['flow_age_us'] >= 0 else row_time_us
            t_s = flow_time_us/1e6
        previous = self._last_sample
        reason = None
        if previous is not None:
            delta_seq = (w['seq']-previous.seq) & 0xffffffff
            if delta_seq != 1:
                reason = 'sequence_discontinuity'
                if 1 < delta_seq < 0x80000000:
                    self.dropped_rows += delta_seq-1
                    if active:
                        self._run_dropped_rows += delta_seq-1
            elif w['scan_id'] is not None and previous.scan_id is not None and (w['scan_id']-previous.scan_id) & 0xffffffff != 1:
                reason = 'scan_discontinuity'
            elif not 0 < (t_s-previous.t_s)*1e6 <= self.peak_config.max_sample_gap_us:
                reason = 'flow_time_discontinuity'
            elif w['missed_slots_total'] != previous.missed_slots_total or w['tx_dropped_total'] != previous.tx_dropped_total:
                reason = 'firmware_loss'
        if not w['flow_ok']:
            reason = 'invalid_flow'
        if reason:
            self._break_locked(reason, w['seq'])
        if w['flow_ok']:
            self._flow_hist.append(w['flow_raw'])
        # Divide before summing so finite extreme inputs cannot poison a
        # running sum. The bounded window is also replayed with this order.
        average = sum(value/len(self._flow_hist) for value in self._flow_hist) if self._flow_hist else 0.0
        resolved = standard_flow(w['target_peak_slm'], self.gas_temperature_c, self.absolute_pressure_mbar)
        target = w['target_peak_slm'] if resolved is None else resolved
        sample = Sample(
            w['seq'], w['ms'], t_s, time.time(), w['flow_raw'], average,
            w['flow_ok'], w['flow_age_us'], w['pa'], w['temp'], w['pok'], w['pa_age_us'],
            w['cmd_dac'], w['feedback_dac'], w['cmd_pct'], w['fdbk_pct'],
            w['target_flow_slm'], target, w['phase'], w['preset'], w['clamped'],
            w['run_id'], w['scan_id'], w['row_us'], w['cycle_id'], w['scan_start_us'],
            w['missed_slots_total'], w['tx_dropped_total'], row_time_us, flow_time_us,
            [row_time_us-a if a >= 0 else None for a in w['pa_age_us']],
            self._segment_id, reason or '', resolved is not None,
            w['target_peak_slm'])
        self._last_seq = sample.seq
        self._last_sample = sample
        self._last_preset = sample.preset
        if active and sample.cycle_id is not None and sample.phase in (0, 1):
            ordinal = sample.cycle_id*2 + sample.phase
            if self._phase_first is None:
                self._phase_first = ordinal
            self._phase_last = ordinal
        return sample

    def _plot_sample(self, sample):
        with self.lock:
            self.t.append(sample.t_s)
            for name in ('flow_raw', 'flow_avg', 'flow_ok', 'flow_age_us', 'cmd_dac',
                         'feedback_dac', 'cmd_pct', 'fdbk_pct', 'target_flow_slm',
                         'target_peak_slm', 'phase', 'preset', 'clamped'):
                value = getattr(sample, name)
                if name in ('flow_raw', 'flow_avg') and not sample.flow_ok:
                    value = float('nan')
                getattr(self, name).append(value)
            self.psens_temp_avg.append(sample.temp)
            for ch in range(7):
                self.psens_pa[ch].append(sample.pa[ch] if sample.pok[ch] else float('nan'))
                self.psens_ok[ch].append(sample.pok[ch])
                self.psens_age_us[ch].append(sample.pa_age_us[ch])

    def snapshot(self):
        with self.lock:
            return (list(self.t), list(self.flow_raw), list(self.flow_avg), list(self.flow_ok),
                    list(self.flow_age_us), [list(x) for x in self.psens_pa], list(self.psens_temp_avg),
                    [list(x) for x in self.psens_ok], [list(x) for x in self.psens_age_us],
                    list(self.cmd_dac), list(self.feedback_dac), list(self.cmd_pct), list(self.fdbk_pct),
                    list(self.target_flow_slm), list(self.target_peak_slm), list(self.phase),
                    list(self.preset), list(self.clamped))

    def _write_peak_row(self, event):
        try:
            self._peak_writer.writerow(event_record(event))
        except OSError:
            raise
        except Exception as exc:
            traceback.print_exc()
            raise RecordingFault('format_fault', exc) from exc
        self._peak_file.flush()
        if event['excursion_complete'] and event['sample'].cycle_id is not None:
            self._completed_phases.add(event['sample'].cycle_id*2+event['sample'].phase)
        if event['excursion_complete'] and self.expected_expiration_sign is not None:
            expected = 1 if self.expected_expiration_sign == 'positive' else -1
            if event['sample'].phase == 1:
                expected *= -1
            if event['direction'] != expected:
                self.orientation_mismatches += 1
                if self.orientation_mismatches == 1:
                    print('-> flow sign disagrees with configured phase orientation')
        result = 'unscored' if event['qualified_peak'] is None else str(event['qualified_peak'])
        print(f"-> peak {event['peak_idx']}: qualified={result}; reasons={','.join(event['reason_codes'])}")

    def _save_manifest(self):
        if self.manifest is None or self._manifest_path is None:
            return
        temporary = self._manifest_path + '.tmp'
        created = False
        try:
            with open(temporary, 'x') as stream:
                created = True
                json.dump(self.manifest, stream, indent=2, allow_nan=False)
                stream.write('\n')
            os.replace(temporary, self._manifest_path)
        finally:
            if created and os.path.exists(temporary):
                os.unlink(temporary)

    def start_recording(self, preset_index=None):
        with self._record_lock:
            if self._record_file is not None:
                return None
            preset = self._last_preset if preset_index is None else preset_index
            base = f"{PRESET_NAMES.get(preset, 'unknown').replace('/', '-').replace(' ', '_')}_{time.strftime('%Y%m%d_%H%M%S')}"
            try:
                paths, files = open_unique_session(RECORDINGS_DIR, base)
                self._record_path, self._peak_path, self._manifest_path = paths
                self._record_file, self._peak_file, manifest_file = files
                manifest_file.close()
                self._record_writer = csv.DictWriter(self._record_file, fieldnames=SAMPLE_COLUMNS)
                self._peak_writer = csv.DictWriter(self._peak_file, fieldnames=PEAK_COLUMNS)
                self._record_writer.writeheader()
                self._peak_writer.writeheader()
                self.manifest = dict(
                    protocol_version=2, sample_schema=2, peak_schema=2, analysis_policy=2,
                    run_id=self.run_id, mode=self.mode, preset=preset,
                    detector_config=asdict(self.peak_config), avg_window=self.avg_window,
                    expected_expiration_sign=self.expected_expiration_sign,
                    flow_reference=dict(gas_temperature_c=self.gas_temperature_c,
                                        absolute_pressure_mbar=self.absolute_pressure_mbar),
                    software_hashes={name: hashlib.sha256(Path(SCRIPT_DIR, name).read_bytes()).hexdigest()
                                     for name in ('Data_Logger.txt', 'peak_detection.py')},
                    firmware=self.metadata, time_origin='STARTED read-completion-relative microseconds',
                    start_ack=None, end_ack=None, boundaries=[], complete=False,
                    tail_truncated=True, termination_reason='open', rows=0,
                    missing_rows=0, invalid_rows=0)
                self.manifest['config_sha256'] = config_fingerprint(self.manifest)
                self._save_manifest()
            except OSError as exc:
                for stream in (self._record_file, self._peak_file):
                    if stream is not None:
                        stream.close()
                self._record_file = self._peak_file = None
                self._record_writer = self._peak_writer = None
                print(f'-> storage fault allocating recording: {exc}')
                return None
            self._record_expected_preset = preset
            self._record_started = False
            self._rows_written = self._run_bad_lines = self._run_dropped_rows = 0
            self.orientation_mismatches = 0
            self._detector = PeakFlowDetector(self.peak_config, self._write_peak_row)
            return paths

    def _phase_summary(self):
        first, last = self._phase_first, self._phase_last
        if first is None or last is None:
            return dict(missing_interior=0, first_phase_partial=False, last_phase_partial=False)
        interior = max(0, last-first-1)
        completed = sum(first < phase < last for phase in self._completed_phases)
        return dict(first_ordinal=first, last_ordinal=last,
                    missing_interior=max(0, interior-completed),
                    first_phase_partial=first not in self._completed_phases,
                    last_phase_partial=last not in self._completed_phases)

    def _finalize_locked(self, reason, tail_truncated):
        if self._record_file is None:
            return
        failed = False
        counters = {}
        try:
            if self._detector is not None:
                self._call_detector_locked('flush')
                counters = {name: getattr(self._detector, name) for name in (
                    'count', 'discarded', 'target_passes', 'target_misses', 'partial_excursions',
                    'warmup_peaks', 'reference_unresolved', 'pressure_incomplete', 'timing_failures',
                    'qualified_peaks', 'unqualified_peaks')}
        except RecordingFault as exc:
            failed = True
            reason = exc.kind
            print(f'-> {exc.kind} finalizing recording: {exc.cause}')
        except OSError as exc:
            failed = True
            reason = 'storage_fault'
            print(f'-> storage fault finalizing recording: {exc}')
        except Exception:
            failed = True
            reason = 'analysis_fault'
            traceback.print_exc()
        # The final manifest is published only after both streams flush/close.
        for stream in (self._record_file, self._peak_file):
            try:
                stream.flush()
                stream.close()
            except OSError as exc:
                failed = True
                reason = 'storage_fault'
                print(f'-> storage fault closing recording: {exc}')
                try:
                    stream.close()
                except OSError:
                    pass
        self.manifest.update(complete=not (tail_truncated or failed),
                             tail_truncated=tail_truncated or failed, termination_reason=reason,
                             rows=self._rows_written, missing_rows=self._run_dropped_rows,
                             invalid_rows=self._run_bad_lines, orientation_mismatches=self.orientation_mismatches,
                             populations=counters, commanded_phases=self._phase_summary())
        try:
            self._save_manifest()
        except OSError as exc:
            print(f'-> storage fault publishing manifest: {exc}; original remains incomplete')
        self._record_file = self._peak_file = None
        self._record_writer = self._peak_writer = None
        self._record_started = False
        self._detector = None
        print(f'-> recording populations (main totals exclude warmup/partial/unresolved): {counters}')

    def _count_bad_line(self):
        self.bad_lines += 1
        fault = None
        with self._record_lock:
            if self._record_started:
                self._run_bad_lines += 1
                try:
                    self._break_locked('malformed_data')
                except RecordingFault as exc:
                    fault = exc
        if fault is not None:
            self._fault(fault.kind, fault.cause)

    def _cancel_run_timer_locked(self):
        self._run_generation += 1
        if self._run_timer is not None:
            self._run_timer.cancel()
        self._run_timer = None
        self._run_start_monotonic = None

    def begin_timed_run(self, preset_index, command):
        with self._run_lock:
            if self.state != 'IDLE' or self.protocol_version != 2 or self.metadata is None:
                print('-> timed runs require idle state and a complete protocol-2 handshake')
                return False
            self.run_id = secrets.token_hex(8)
            self.mode = 'DEMO' if preset_index == -2 else f'P{preset_index}'
            self._run_generation += 1
            self.state = 'STARTING'
            self._started_event = threading.Event()
            self._ended_event = threading.Event()
            started_event = self._started_event
            owned_id = self.run_id
            deadline = time.monotonic() + 2.0
            if self.start_recording(preset_index) is None:
                self.state = 'IDLE'
                return False
            with self._serial_write_lock:
                self._heartbeat_enabled = True
                sent = self._write_bytes_locked(
                    f'@START,{self.run_id},{self.mode}\n'.encode('ascii'),
                    timeout=max(.001, deadline-time.monotonic()))
        # Reader takes both locks for ACK; no lifecycle/record lock while waiting.
        if not sent or not started_event.wait(max(0, deadline-time.monotonic())):
            self._fault('start_unconfirmed', expected_run_id=owned_id)
            return False
        return True

    def _on_run_timeout(self, generation):
        self.stop_run(generation=generation)

    def stop_run(self, generation=None, discard=False):
        with self._run_lock:
            if generation is not None and generation != self._run_generation:
                return False
            if self.state not in ('STARTING', 'RUNNING', 'STOPPING'):
                return False
            owner = self.state != 'STOPPING'
            owned_id = self.run_id
            ended_event = self._ended_event
            paths = (self._record_path, self._peak_path, self._manifest_path)
            if owner:
                self.state = 'STOPPING'
                self._cancel_run_timer_locked()
                self._stop_deadline = time.monotonic()+2.0
            deadline = self._stop_deadline
            with self._serial_write_lock:
                self._heartbeat_enabled = False
                if owner:
                    sent = self._write_bytes_locked(
                        f'@STOP,{owned_id}\n'.encode('ascii'),
                        timeout=max(.001, deadline-time.monotonic()))
                else:
                    sent = True
        confirmed = sent and ended_event.wait(min(1.0, max(0, deadline-time.monotonic())))
        if not confirmed and owner and sent:
            with self._run_lock:
                if self.run_id == owned_id and self.state == 'STOPPING' and time.monotonic() < deadline:
                    with self._serial_write_lock:
                        sent = self._write_bytes_locked(
                            f'@STOP,{owned_id}\n'.encode('ascii'),
                            timeout=max(.001, deadline-time.monotonic()))
            confirmed = sent and ended_event.wait(max(0, deadline-time.monotonic()))
        elif not confirmed and sent:
            confirmed = ended_event.wait(max(0, deadline-time.monotonic()))
        if not confirmed:
            self._fault('stop_unconfirmed', expected_run_id=owned_id)
        if discard:
            self._discard_paths(paths)
        return confirmed

    def _fault(self, reason, exc=None, expected_run_id=None):
        with self._run_lock:
            if expected_run_id is not None and (self.run_id != expected_run_id or self.state == 'IDLE'):
                return
            with self._serial_write_lock:
                self._heartbeat_enabled = False
            self._cancel_run_timer_locked()
            self.state = 'FAULTED'
            self.protocol_version = None
            self.metadata = None
            self._meta_pending = None
            self._meta_build = None
            self._meta_invalid = True
            self._ready_event.clear()
            with self._record_lock:
                self._finalize_locked(reason, True)
        print(f'-> {reason}: {exc or "stop unconfirmed; heartbeat disabled"}')

    def run_in_progress(self):
        with self._run_lock:
            return self.state in ('STARTING', 'RUNNING', 'STOPPING')

    def run_remaining(self):
        with self._run_lock:
            if self._run_start_monotonic is None:
                return None
            return max(0.0, self.run_seconds-(time.monotonic()-self._run_start_monotonic))

    def end_run_override(self):
        return self.stop_run()

    def abort_run(self):
        return self.stop_run(discard=True)

    def stop_recording(self):
        return self.stop_run()

    @staticmethod
    def _discard_paths(paths):
        for path in paths:
            if path is not None:
                try:
                    os.unlink(path)
                except FileNotFoundError:
                    pass
                except OSError as exc:
                    print(f'-> could not remove recording: {exc}')

    def close(self):
        if self.run_in_progress():
            self.stop_run()
        self._stop.set()
        with self._serial_write_lock:
            self._heartbeat_enabled = False
            self.ser.close()
        if self._thread.ident is not None and self._thread is not threading.current_thread():
            self._thread.join(timeout=2.0)

def visible_start(t, window_s):
    """Index of the first sample within `window_s` seconds of the latest one.

    Binary search rather than the previous linear scan: this runs three
    times per frame over a buffer that can hold tens of thousands of
    samples, and it was the second-largest cost in the redraw path.
    """
    if not t:
        return 0
    return bisect.bisect_left(t, max(0.0, t[-1] - window_s))


def autoscale(ax, values, min_span=1.0, pad_frac=0.15):
    """Pad around the visible min/max, never collapsing to a zero-height
    window when the signal is flat."""
    values = [v for v in values if math.isfinite(v)]
    if not values:
        return
    lo, hi = min(values), max(values)
    pad = max((hi - lo) * pad_frac, min_span)
    ax.set_ylim(lo - pad, hi + pad)


def set_title(fig, text):
    try:
        fig.canvas.manager.set_window_title(text)
    except Exception:
        pass  # backend-dependent; purely cosmetic


def positive_int(value):
    try:
        parsed = int(value)
    except ValueError as exc:
        raise argparse.ArgumentTypeError("must be an integer") from exc
    if parsed <= 0:
        raise argparse.ArgumentTypeError("must be greater than zero")
    return parsed


def nonnegative_int(value):
    try:
        parsed = int(value)
    except ValueError as exc:
        raise argparse.ArgumentTypeError("must be an integer") from exc
    if parsed < 0:
        raise argparse.ArgumentTypeError("must be zero or greater")
    return parsed


def positive_float(value):
    try:
        parsed = float(value)
    except ValueError as exc:
        raise argparse.ArgumentTypeError("must be a number") from exc
    if not math.isfinite(parsed) or not (parsed > 0.0):
        raise argparse.ArgumentTypeError("must be greater than zero")
    return parsed


def unit_fraction(value):
    parsed = positive_float(value)
    if parsed > 1.0:
        raise argparse.ArgumentTypeError("must be no greater than one")
    return parsed


def percentage(value):
    parsed = positive_float(value)
    if parsed > 100.0:
        raise argparse.ArgumentTypeError("must be no greater than 100")
    return parsed

                    help="Centred moving-average window used for offline peak "
                         "location. Independent default: 9; must be odd.")
    ap.add_argument("--buffer", type=positive_float, default=180.0,
                    help="Seconds of history retained in memory.")
    ap.add_argument("--peak-threshold-frac", type=unit_fraction,
                    default=DEFAULT_PEAK_THRESHOLD_FRAC,
                    help="Fraction of the current phase target used to open an "
                         "excursion (default: 0.35).")
    ap.add_argument("--peak-threshold-floor", type=positive_float,
                    default=DEFAULT_PEAK_THRESHOLD_FLOOR,
                    help="Minimum excursion-opening threshold in slm, also used "
                         "when no phase target is available (default: 1.5).")
    ap.add_argument("--peak-exit", type=unit_fraction, default=0.5,
                    help="Excursion closes when |flow| drops below this fraction "
                         "of its latched adaptive threshold.")
    ap.add_argument("--peak-min-samples", type=positive_int, default=9,
                    help="Excursions shorter than this are discarded as noise.")
    ap.add_argument("--peak-plateau", type=unit_fraction, default=0.99,
                    help="Plateau = samples within this fraction of the peak; the "
                         "peaks CSV reports the mean pressure across it. The 0.99 "
                         "default limits Q-squared pressure variation to about 2%%.")
    ap.add_argument("--warmup-breaths", type=nonnegative_int,
                    default=DEFAULT_WARMUP_BREATHS,
                    help="Initial complete breath cycles marked is_warmup in the "
                         "peaks CSV (default: 3; rows are retained).")
    ap.add_argument("--peak-tolerance-pct", type=percentage,
                    default=DEFAULT_PEAK_TOLERANCE_PCT,
                    help="Allowed absolute error from the phase's commanded "
                         "peak flow before the peaks CSV marks a miss "
                         "(default: 10%%).")
    ap.add_argument("--max-sensor-age-ms", type=positive_float,
                    default=DEFAULT_MAX_SENSOR_AGE_MS,
                    help="Maximum flow/pressure sample age at a detected peak "
                         "before sensor_timing_ok is false (default: 20 ms).")
    ap.add_argument("--run-seconds", type=positive_float,
                    default=DEFAULT_RUN_SECONDS,
                    help="Duration of each preset's automatic recording before "
                         "the actuator returns to hold (default: 180).")
    ap.add_argument("--list-ports", action="store_true", help="List serial ports and exit.")
    ap.add_argument('--gas-temperature-c', type=float)
    ap.add_argument('--absolute-pressure-mbar', type=float)
    ap.add_argument('--expected-expiration-sign', choices=('positive', 'negative'))
    ap.add_argument('--max-sample-gap-us', type=positive_float, default=15000)
    args = ap.parse_args()
    try:
        standard_flow(1.0, args.gas_temperature_c, args.absolute_pressure_mbar)
    except ValueError as exc:
        ap.error(str(exc))

    if args.list_ports:
        list_ports()
        return

    peak_config = PeakDetectionConfig(
        threshold_frac=args.peak_threshold_frac,
        min_threshold=args.peak_threshold_floor,
        exit_frac=args.peak_exit,
        min_samples=args.peak_min_samples,
        plateau_frac=args.peak_plateau,
        smooth_window=args.peak_smooth,
        target_tolerance_frac=args.peak_tolerance_pct / 100.0,
        max_sensor_age_us=args.max_sensor_age_ms * 1000.0,
        warmup_breaths=args.warmup_breaths,
        max_sample_gap_us=args.max_sample_gap_us,
    )
    port = args.port or pick_port()
    print(f"Connecting to {port} @ {args.baud}...")
    try:
        link = SerialLink(port, args.baud, args.avg,
                          buffer_seconds=args.buffer,
                          peak_config=peak_config,
                          run_seconds=args.run_seconds,
                          gas_temperature_c=args.gas_temperature_c,
                          absolute_pressure_mbar=args.absolute_pressure_mbar,
                          expected_expiration_sign=args.expected_expiration_sign)
    except (serial.SerialException, OSError) as e:
        sys.exit(f"Couldn't open {port}: {e}\n"
                 f"Close the Arduino Serial Monitor (only one program can hold "
                 f"the port) and check the board is plugged in.")

    # Native USB port opening may not reset the board. Wait through normal
    # centre-hold, then require an acknowledgement emitted only from loop().
    time.sleep(2.5)
    if not link.wait_until_ready(5.0):
        link.close()
        sys.exit("Firmware did not acknowledge readiness within 5 seconds; "
                 "the board may still be booting or running an incompatible "
                 "sketch.")
    print(f"Connected. {COMMAND_HELP}\n")

    # Free up the keys we use from matplotlib's default keymaps (h/s/q/etc.
    # are normally bound to home/save/back/quit) so on_key() is the only
    # thing that responds to them.
    for action in ("keymap.home", "keymap.back", "keymap.forward", "keymap.save",
                   "keymap.quit", "keymap.grid", "keymap.yscale", "keymap.xscale",
                   "keymap.pan", "keymap.zoom", "keymap.fullscreen", "keymap.all_axes",
                   "keymap.help", "keymap.copy"):
        if action in plt.rcParams:  # keymap entries vary across matplotlib versions
            plt.rcParams[action] = []

    def on_key(event):
        key = event.key
        if key in PRESET_KEYS:
            if link.run_in_progress():
                print("-> a timed run is already active; press 'p' to abort "
                      "or H to keep and end it")
                return
            if link.begin_timed_run(int(key) - 1, key):
                print(f"-> sent '{key}'  ({COMMANDS[key]})")
        elif key in SAFETY_OVERRIDE_KEYS or key in MOTION_OVERRIDE_KEYS:
            link.send_command(key)
        elif key in COMMANDS:
            if link.send_command(key):
                print(f"-> sent '{key}'  ({COMMANDS[key]})")
        elif event.key == "p":
            link.abort_run()
        elif event.key == "q":
            plt.close("all")

    # ---------------- Figure 1: flow sensor ----------------
    fig_flow, ax_flow = plt.subplots(figsize=(8, 4.5))
    set_title(fig_flow, "Flow sensor — live")
    fig_flow.suptitle(COMMAND_HELP, fontsize=8)
    raw_line, = ax_flow.plot([], [], lw=1, color="#a9c6e8", alpha=0.6, label="raw")
    avg_line, = ax_flow.plot([], [], lw=2, color="#1f6feb", label=f"smoothed (avg {args.avg})")
    target_line, = ax_flow.plot([], [], lw=1.2, ls="--", color="#7b1fa2",
                                label="geometric target (local L/min)")
    # Adaptive opening/exit levels follow the current phase target. The peak
    # detector latches them per excursion; these lines are a live preview.
    threshold_guides = []
    for multiplier, style in ((1.0, "--"), (peak_config.exit_frac, ":")):
        for sign in (1, -1):
            guide = ax_flow.axhline(0, color="#c62828", lw=0.7,
                                    ls=style, alpha=0.6)
            threshold_guides.append((guide, multiplier, sign))
    ax_flow.axhline(0, color="#999", lw=0.8)
    ax_flow.set_xlabel("Time (s)")
    ax_flow.set_ylabel("Flow (slm)")
    ax_flow.grid(alpha=0.25)
    ax_flow.legend(loc="upper right", fontsize=8)
    flow_status = ax_flow.text(0.01, 0.97, "", transform=ax_flow.transAxes,
                               va="top", fontsize=9, family="monospace")
    fig_flow.canvas.mpl_connect("key_press_event", on_key)

    # ---------------- Figure 2: actuator command/feedback ----------------
    fig_act, ax_act = plt.subplots(figsize=(8, 4.5))
    set_title(fig_act, "Actuator command / feedback — live")
    fig_act.suptitle(COMMAND_HELP, fontsize=8)
    cmd_line, = ax_act.plot([], [], lw=1.5, color="#e07b39", label="commanded (normalized DAC)")
    fdbk_line, = ax_act.plot([], [], lw=1.5, color="#2ca02c", label="feedback (normalized DAC)")
    ax_act.axhline(0, color="#999", lw=0.8)
    ax_act.set_ylim(-115, 115)
    ax_act.set_xlabel("Time (s)")
    ax_act.set_ylabel("Position (100 × DAC offset / half-range)")
    ax_act.grid(alpha=0.25)
    ax_act.legend(loc="upper right", fontsize=8)
    act_status = ax_act.text(0.01, 0.97, "", transform=ax_act.transAxes,
                             va="top", fontsize=9, family="monospace")
    fig_act.canvas.mpl_connect("key_press_event", on_key)

    # ---------------- Figure 3: pressure sensors (7x SDP811) ----------------
    fig_press, ax_press = plt.subplots(figsize=(8, 4.5))
    set_title(fig_press, "Pressure sensors — live")
    fig_press.suptitle(COMMAND_HELP, fontsize=8)
    press_colors = plt.cm.tab10.colors  # 10 distinct colors, first 7 used
    press_lines = []
    for i in range(NUM_PSENS):
        (line,) = ax_press.plot([], [], lw=1.3, color=press_colors[i], label=f"Psens{i + 1}")
        press_lines.append(line)
    ax_press.axhline(0, color="#999", lw=0.8)
    ax_press.set_xlabel("Time (s)")
    ax_press.set_ylabel("Differential pressure (Pa)")
    ax_press.grid(alpha=0.25)
    ax_press.legend(loc="upper right", fontsize=7, ncol=4)
    press_status = ax_press.text(0.01, 0.97, "", transform=ax_press.transAxes,
                                 va="top", fontsize=9, family="monospace")
    fig_press.canvas.mpl_connect("key_press_event", on_key)

    def set_xwindow(ax, t):
        t_now = t[-1]
        t_lo = max(0.0, t_now - args.window)
        ax.set_xlim(t_lo, max(t_now, t_lo + 1.0))

    def update_flow(_frame):
        (t, flow_raw, flow_avg, flow_ok, flow_age_us, *_middle,
         target_flow, target_peak, phase, _preset, _clamped) = link.snapshot()
        if not t:
            return raw_line, avg_line, target_line
        i0 = visible_start(t, args.window)
        tw, raw_w, avg_w = t[i0:], flow_raw[i0:], flow_avg[i0:]
        target_w = target_flow[i0:]
        raw_line.set_data(tw, raw_w)
        avg_line.set_data(tw, avg_w)
        target_line.set_data(tw, target_w)
        current_threshold = peak_config.opening_threshold(target_peak[-1])
        for guide, multiplier, sign in threshold_guides:
            guide.set_ydata([sign * current_threshold * multiplier] * 2)
        set_xwindow(ax_flow, t)
        autoscale(ax_flow, raw_w + avg_w + target_w)
        remaining = link.run_remaining()
        with link._record_lock:
            recording = link._record_file is not None
        rec = (f" [RUN {remaining:5.1f}s]" if remaining is not None else
               (" [REC]" if recording else ""))
        link_state = ("CONNECTED" if link.connected else
                      f"DISCONNECTED: {link.connection_error or 'reader stopped'}")
        flow_status.set_text(
            f"flow: {flow_avg[-1]:7.2f} slm   "
            f"geometric target: {target_flow[-1]:7.2f} local L/min   "
            f"threshold: {current_threshold:5.2f} slm   "
            f"phase: {PHASE_NAMES.get(phase[-1], phase[-1])}\n"
            f"[{'OK' if flow_ok[-1] else 'SENSOR FAULT'}; "
            f"age {flow_age_us[-1] / 1000.0:.2f} ms]{rec}\n{link_state}"
        )
        return raw_line, avg_line, target_line

    def update_act(_frame):
        (t, *_rest, cmd_dac, feedback_dac, cmd_pct, fdbk_pct,
         _target_flow, _target_peak, _phase, preset, clamped) = link.snapshot()
        if not t:
            return cmd_line, fdbk_line
        i0 = visible_start(t, args.window)
        cmd_line.set_data(t[i0:], cmd_pct[i0:])
        fdbk_line.set_data(t[i0:], fdbk_pct[i0:])
        set_xwindow(ax_act, t)
        act_status.set_text(
            f"cmd: {cmd_pct[-1]:6.1f}%   fdbk: {fdbk_pct[-1]:6.1f}%   "
            f"({cmd_dac[-1]:4d}/{feedback_dac[-1]:7.2f} DAC)   "
            f"mode: {PRESET_NAMES.get(preset[-1], preset[-1])}"
            f"{'   [CLAMPED]' if clamped[-1] else ''}"
        )
        return cmd_line, fdbk_line

    def update_press(_frame):
        (t, _fr, _fa, _fo, _flow_age, psens_pa, psens_temp_avg,
         psens_ok, psens_age_us, *_) = link.snapshot()
        if not t:
            return tuple(press_lines)
        i0 = visible_start(t, args.window)
        tw = t[i0:]
        all_vals = []
        for i, pline in enumerate(press_lines):
            yw = psens_pa[i][i0:]
            pline.set_data(tw, yw)
            all_vals.extend(yw)
        set_xwindow(ax_press, t)
        autoscale(ax_press, all_vals)
        ok_count = sum(1 for i in range(NUM_PSENS) if psens_ok[i][-1])
        valid_ages = [ages[-1] for ages, ok in zip(psens_age_us, psens_ok)
                      if ok[-1] and ages[-1] >= 0]
        age_text = (f"{max(valid_ages) / 1000.0:.2f} ms"
                    if valid_ages else "unavailable")
        press_status.set_text(
            f"sensors ok: {ok_count}/{NUM_PSENS}   "
            f"avg temp: {psens_temp_avg[-1]:5.1f} C   "
            f"max valid age: {age_text}"
        )
        return tuple(press_lines)

    anims = [
        animation.FuncAnimation(fig_flow, update_flow, interval=80,
                                blit=False, cache_frame_data=False),
        animation.FuncAnimation(fig_act, update_act, interval=80,
                                blit=False, cache_frame_data=False),
        animation.FuncAnimation(fig_press, update_press, interval=80,
                                blit=False, cache_frame_data=False),
    ]

    for fig in (fig_flow, fig_act, fig_press):
        fig.tight_layout(rect=(0, 0, 1, 0.94))

    try:
        plt.show()  # blocks until all three windows are closed
    except KeyboardInterrupt:
        pass
    finally:
        del anims
        link.close()


if __name__ == "__main__":
    main()
